% ヘッダ文字の書式：ラベル部分（「定義3.2」など）をチップ状に囲む
% 演習だけ黒背景・白文字で強調し，それ以外は共通のグレーチップにする
\newtheoremstyle{logicchip}%
	{6pt}{4pt}%
	{\normalfont}%
	{0pt}%
	{\normalfont\bfseries\headfont}{}{.5em}%
	{\colorbox{black!8}{\thmname{#1}\thmnumber{\,#2}}\thmnote{\,(#3)}}
\newtheoremstyle{logicque}%
	{5pt}{2pt}%
	{\normalfont}%
	{0pt}%
	{\normalfont\bfseries\headfont}{}{.5em}%
	{\colorbox{black}{\textcolor{white}{\thmname{#1}\thmnumber{\,#2}}}\thmnote{\,(#3)}}
\theoremstyle{logicchip}
\newtheorem{Def}{定義}[section]
\newtheorem{Thm}[Def]{定理}
\newtheorem{Ex}[Def]{例}
\theoremstyle{logicque}
\newtheorem{Que}{演習}[section]
\theoremstyle{logicchip}
\newtheorem{Lemma}[Def]{補題}
\newtheorem{Corollary}[Def]{系}
\newtheorem{Note}[Def]{注意}

}

% 節・小節見出し：左に縦アクセントバーのみ（下線なし）．
\makeatletter
\renewcommand{\section}{%
	\if@noskipsec \leavevmode \fi
	\par
	\addpenalty\@secpenalty
	\addvspace{\Cvs \@plus.5\Cdp \@minus.2\Cdp}%
	\@afterheading
	\@ifstar{\@logicssect}{\@logicsect}}
\def\@logicsect#1{%
	\ifnum 1>\c@secnumdepth
		\let\@logicsvsec\@empty
	\else
		\refstepcounter{section}%
		\protected@edef\@logicsvsec{\presectionname\thesection\quad}%
	\fi
	\begingroup
		\normalfont\Large\headfont\raggedright
		\rule[-.28\baselineskip]{3pt}{\baselineskip}\hspace{14pt}%
		\@logicsvsec
		\interlinepenalty\@M #1\@@par
	\endgroup
	\sectionmark{#1}%
	\addcontentsline{toc}{section}{%
		\ifnum 1>\c@secnumdepth\else\protect\numberline{\thesection}\fi #1}%
	\@xsect{.5\Cvs \@plus.3\Cdp}}
\def\@logicssect#1{%
	\begingroup
		\normalfont\Large\headfont\raggedright
		\rule[-.28\baselineskip]{3pt}{\baselineskip}\hspace{14pt}%
		\interlinepenalty\@M #1\@@par
	\endgroup
	\@xsect{.5\Cvs \@plus.3\Cdp}}
%
\renewcommand{\subsection}{%
	\if@noskipsec \leavevmode \fi
	\par
	\addpenalty\@secpenalty
	\addvspace{\Cvs \@plus.5\Cdp \@minus.2\Cdp}%
	\@afterheading
	\@ifstar{\@logicssubsect}{\@logicsubsect}}
\def\@logicsubsect#1{%
	\ifnum 2>\c@secnumdepth
		\let\@logicsvsubsec\@empty
	\else
		\refstepcounter{subsection}%
		\protected@edef\@logicsvsubsec{\presubsectionname\thesubsection\quad}%
	\fi
	\begingroup
		\normalfont\large\headfont\raggedright
		\color{black!55}\rule[-.24\baselineskip]{2pt}{\baselineskip}\hspace{12pt}%
		\color{black}\@logicsvsubsec
		\interlinepenalty\@M #1\@@par
	\endgroup
	\addcontentsline{toc}{subsection}{%
		\ifnum 2>\c@secnumdepth\else\protect\numberline{\thesubsection}\fi #1}%
	\@xsect{.5\Cvs \@plus.3\Cdp}}

	\begingroup
		\normalfont\large\headfont\raggedright
		\color{black!55}\rule[-.24\baselineskip]{2pt}{\baselineskip}\hspace{12pt}%
		\color{black}%
		\interlinepenalty\@M #1\@@par
	\endgroup
	\@xsect{.5\Cvs \@plus.3\Cdp}}
